02 Pointing the Mount

Mount + tracking

A motor spins one tilted axis at the sky's own rate, so a chosen star stays pinned while the Earth turns under it.

Drift
Earth rotates ~15°/hr; uncorrected, every star slides across the frame.
Fix
An equatorial mount tilts one axis parallel to Earth's; a motor turns it at sidereal rate to cancel the drift.
Coords
RA = sky longitude, Dec = sky latitude. A go-to maps a target to a mount move.
Reality
Tracking is close but imperfect — good for short subs, not flawless for long ones.
locked target tracking: off

Without tracking a multi-second exposure smears every star into a streak; the longer the sub, the worse the trail.

AstroDeck's mount panel: slew (move now), track (hold sidereal), go-to (point at a named target).

You are bolting a camera to a clock and asking it to undo the planet.

SPEED * TRACK_AT : SPEED * elapsed; ctx.clearRect(0, 0, W, H); ctx.fillStyle = "#070910"; ctx.fillRect(0, 0, W, H); // faint center crosshair where the target should sit ctx.strokeStyle = LINE; ctx.lineWidth = 1; ctx.beginPath(); ctx.moveTo(target.x - 14, target.y); ctx.lineTo(target.x + 14, target.y); ctx.moveTo(target.x, target.y - 14); ctx.lineTo(target.x, target.y + 14); ctx.stroke(); // background stars: wrap horizontally as they drift for (var k = 0; k < stars.length; k++) { var s = stars[k]; var x = ((s.x - off) % W + W) % W; ctx.globalAlpha = s.a; ctx.fillStyle = INK; ctx.beginPath(); ctx.arc(x, s.y, s.r, 0, Math.PI * 2); ctx.fill(); } ctx.globalAlpha = 1; // the target star: drifts off-center while untracked, snaps and // holds center once tracking is on. var tx; if (tracking) { // ease toward center over ~0.5s after lock var p = Math.min(1, (elapsed - TRACK_AT) / 0.5); var drifted = ((target.x - SPEED * TRACK_AT) % W + W) % W; tx = drifted + (target.x - drifted) * (p * p * (3 - 2 * p)); } else { tx = ((target.x - off) % W + W) % W; } // glow var glow = ctx.createRadialGradient(tx, target.y, 0, tx, target.y, 12); glow.addColorStop(0, "rgba(232,168,73,0.55)"); glow.addColorStop(1, "rgba(232,168,73,0)"); ctx.fillStyle = glow; ctx.beginPath(); ctx.arc(tx, target.y, 12, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = AMBER; ctx.beginPath(); ctx.arc(tx, target.y, target.r, 0, Math.PI * 2); ctx.fill(); if (label) label.textContent = tracking ? "on — locked" : "off — drifting"; } function frame(now) { if (!t0) t0 = now; var elapsed = (now - t0) / 1000; if (elapsed > LOOP_AT) { t0 = now; elapsed = 0; } // loop the demo paint(elapsed); raf = requestAnimationFrame(frame); } window.AstroAnim.register(cv, { play: function () { cancelAnimationFrame(raf); if (window.AstroAnim.reduceMotion) { // Reduced motion: paint the meaningful final frame (locked). paint(LOOP_AT - 0.01); return; } t0 = 0; raf = requestAnimationFrame(frame); }, reset: function () { cancelAnimationFrame(raf); t0 = 0; paint(0); // back to frame zero: drifting, target off start } }); })();
